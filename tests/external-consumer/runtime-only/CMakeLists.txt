cmake_minimum_required(VERSION 3.16)
project(cog_installed_runtime C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(cog-provider REQUIRED CONFIG)
find_package(Threads REQUIRED)
foreach(component config toolkit orchestrator)
 include("${cog-provider_DIR}/../cog/cog-${component}-targets.cmake")
endforeach()
include("${cog-provider_DIR}/../cog-instructions/cog-instructions-targets.cmake")
include("${cog-provider_DIR}/../cog-runtime/cog-runtime-targets.cmake")
if(TARGET cog::cli OR TARGET cog::cli-config)
 message(FATAL_ERROR "Runtime consumer must not import CLI targets")
endif()
add_executable(installed_runtime consumer.c)
target_link_libraries(installed_runtime PRIVATE cog::runtime)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(installed_runtime PRIVATE -Wall -Wextra -Wpedantic -Wconversion
  -Wsign-conversion -Wshadow -Wstrict-prototypes -Wmissing-prototypes -Werror)
endif()
