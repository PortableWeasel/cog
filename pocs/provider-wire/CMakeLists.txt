cmake_minimum_required(VERSION 3.16)
project(provider-wire-research C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(WIRE_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_library(research-json STATIC vendor/yyjson.c)
target_include_directories(research-json SYSTEM PUBLIC vendor)
add_executable(provider-wire wire.c)
target_link_libraries(provider-wire PRIVATE research-json)
target_compile_options(provider-wire PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Wstrict-prototypes -Wmissing-prototypes -Werror)
if(WIRE_SANITIZE)
  foreach(t research-json provider-wire)
    target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endforeach()
  target_link_options(provider-wire PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME wire-contract COMMAND provider-wire)
add_test(NAME wire-live-tool-fixture COMMAND provider-wire ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/live-tool.sse tool)
add_test(NAME wire-live-answer-fixture COMMAND provider-wire ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/live-answer.sse text)
