cmake_minimum_required(VERSION 3.18)
project(cog_https_lifecycle_poc LANGUAGES C)
# Deliberately standalone: never included by Cog's root CMake build.
find_package(PkgConfig REQUIRED)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl>=7.85.0)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_program(OPENSSL_EXECUTABLE openssl REQUIRED)
option(POC_SANITIZERS "Enable address/undefined sanitizers" OFF)
add_executable(https-lifecycle lifecycle.c)
target_compile_features(https-lifecycle PRIVATE c_std_11)
set_target_properties(https-lifecycle PROPERTIES C_EXTENSIONS OFF)
target_compile_options(https-lifecycle PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Wsign-conversion -Wmissing-prototypes -Wstrict-prototypes -Werror)
target_link_libraries(https-lifecycle PRIVATE PkgConfig::CURL)
if(POC_SANITIZERS)
  target_compile_options(https-lifecycle PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(https-lifecycle PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME https-lifecycle COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/fixtures.py" "$<TARGET_FILE:https-lifecycle>" "${OPENSSL_EXECUTABLE}")
set_tests_properties(https-lifecycle PROPERTIES TIMEOUT 45)
