foreach(name instructions boundaries)
    add_executable(test-instruction-${name} test_${name}.c)
    target_link_libraries(test-instruction-${name} PRIVATE cog::instructions)
    if(COMMAND target_enable_warnings)
        target_enable_warnings(test-instruction-${name})
    endif()
    add_test(NAME instruction-${name} COMMAND test-instruction-${name})
    set_tests_properties(instruction-${name} PROPERTIES LABELS "instructions;contract")
endforeach()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    add_executable(test-instruction-codec test_codec.c)
    target_link_libraries(test-instruction-codec PRIVATE cog::instructions)
    target_compile_options(test-instruction-codec PRIVATE -fno-builtin)
    target_link_options(test-instruction-codec PRIVATE
        -Wl,--wrap=malloc,--wrap=calloc,--wrap=realloc)
    if(COMMAND target_enable_warnings)
        target_enable_warnings(test-instruction-codec)
    endif()
    add_test(NAME instruction-codec COMMAND test-instruction-codec)
    set_tests_properties(instruction-codec PROPERTIES LABELS "instructions;contract")
    add_executable(test-instruction-allocations test_allocations.c)
    target_link_libraries(test-instruction-allocations PRIVATE cog::instructions)
    target_link_options(test-instruction-allocations PRIVATE
        -Wl,--wrap=malloc,--wrap=calloc,--wrap=realloc)
    if(COMMAND target_enable_warnings)
        target_enable_warnings(test-instruction-allocations)
    endif()
    add_test(NAME instruction-allocations COMMAND test-instruction-allocations)
    set_tests_properties(instruction-allocations PROPERTIES LABELS "instructions;contract")
endif()
